// CAP-04 test-only entry point. Query failure prevents mutation.
PROC
PROC_AESN_TestApplyOneHp((CHARACTER)_Target)
AND
GetHitpoints(_Target, _BeforeCurrent)
AND
GetMaxHitpoints(_Target, _BeforeMaximum)
AND
GetHitpointsPercentage(_Target, _BeforePercentage)
AND
_BeforeCurrent > 0
AND
NOT DB_AESN_TestPendingApply(_Target, _, _, _)
THEN
DB_AESN_TestPendingApply(_Target, _BeforeCurrent, _BeforeMaximum, _BeforePercentage);
ApplyStatus(_Target, "AESN_HP_BIT_00001", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);

// Restore the captured percentage exactly once for a living target after acknowledgement.
IF
StatusApplied((CHARACTER)_Target, "AESN_HP_BIT_00001", _, _)
AND
DB_AESN_TestPendingApply(_Target, _BeforeCurrent, _BeforeMaximum, _BeforePercentage)
AND
_BeforeCurrent > 0
AND
GetMaxHitpoints(_Target, _AfterMaximum)
THEN
SetHitpointsPercentage(_Target, _BeforePercentage, "Guaranteed");
DB_AESN_TestObservation("APPLY", _Target, _BeforeCurrent, _BeforeMaximum, _AfterMaximum, 1, _BeforePercentage);
NOT DB_AESN_TestPendingApply(_Target, _BeforeCurrent, _BeforeMaximum, _BeforePercentage);

// Observe the post-restoration state independently of the acknowledgement rule.
IF
DB_AESN_TestObservation("APPLY", _Target, _, _, _, 1, _)
AND
GetHitpoints(_Target, _AfterCurrent)
AND
GetMaxHitpoints(_Target, _AfterMaximum)
AND
GetHitpointsPercentage(_Target, _AfterPercentage)
THEN
DB_AESN_TestObservation("APPLY_POST", _Target, _AfterCurrent, _AfterMaximum, _AfterMaximum, 1, _AfterPercentage);

// Emit a main-log record only when the exact one-bit and percentage invariants pass.
IF
DB_AESN_TestObservation("APPLY", (CHARACTER)_Target, _BeforeCurrent, _BeforeMaximum, _AppliedMaximum, 1, _BeforePercentage)
AND
DB_AESN_TestObservation("APPLY_POST", _Target, _AfterCurrent, _AfterMaximum, _AfterMaximum, 1, _AfterPercentage)
AND
IntegerSum(_BeforeMaximum, 1, _ExpectedMaximum)
AND
_AppliedMaximum == _ExpectedMaximum
AND
_AfterMaximum == _ExpectedMaximum
AND
_AfterPercentage == _BeforePercentage
AND
ConcatenateInteger("AESN_CAP04 APPLY beforeCurrent=", _BeforeCurrent, _Message1)
AND

AND
ConcatenateInteger(_Message2, _BeforeMaximum, _Message3)
AND
Concatenate(_Message3, ",afterCurrent=", _Message4)
AND
ConcatenateInteger(_Message4, _AfterCurrent, _Message5)
AND

AND
