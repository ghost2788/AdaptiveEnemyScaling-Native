// Bit membership is computed independently by quotient parity. Because delta
// is bounded to 0..65535, the 16 calls above form an exact binary registry.
PROC
PROC_AESN_QueueHpBit((GUIDSTRING)_Combat, (CHARACTER)_Enemy, (INTEGER)_Delta, (INTEGER)_Bit, (STRING)_Status)
AND
IntegerDivide(_Delta, _Bit, _Quotient)
AND
IntegerDivide(_Quotient, 2, _Half)
AND
IntegerProduct(_Half, 2, _Even)
AND
IntegerSubtract(_Quotient, _Even, 1)
THEN
DB_AESN_HpDesiredBit(_Combat, _Enemy, _Bit, _Status);
