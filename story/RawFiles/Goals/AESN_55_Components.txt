// Hardened application starts only after exact HP commit. The frozen schema-2
// tier selects exactly one visible, mutually exclusive status. Relentless is
// deliberately absent from this production path until its runtime proof gate
// is satisfied.
IF
DB_AESN_HpTransaction(_Combat, _Enemy, _Representation, "HPCommitted", _, _, _, _, _Delta, _Delta)
AND
QRY_AESN_HpRepresentationSupported(_Representation)
AND
NOT DB_AESN_HpApplicationHold(_Combat, _Enemy)
AND
DB_AESN_CombatSnapshotV2(_Combat, 2, _, _, _, _, _HardenedTier, _, _, _, _, "Supported")
AND
NOT DB_AESN_ComponentStarted(_Combat, _Enemy)
AND
NOT DB_AESN_ComponentApplication(_Combat, _Enemy, _, _)
AND
NOT DB_AESN_MergeInProgress(_, _Combat)
AND
NOT DB_AESN_MergeInProgress(_Combat, _)
THEN
DB_AESN_ComponentStarted(_Combat, _Enemy);
DB_AESN_ComponentApplication(_Combat, _Enemy, 1, "ApplyingStat");
PROC_AESN_ApplyHardenedStatus(_Combat, _Enemy, _HardenedTier);

PROC
PROC_AESN_ApplyHardenedStatus((GUIDSTRING)_Combat, (CHARACTER)_Enemy, (INTEGER)_HardenedTier)
AND
_HardenedTier == 1
THEN
DB_AESN_ComponentPendingApply(_Combat, _Enemy, "Stat", "AESN_HARDENED_FOE_01");
ApplyStatus(_Enemy, "AESN_HARDENED_FOE_01", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);
RealtimeObjectTimerLaunch(_Enemy, "AESN_COMPONENT_APPLY_TIMEOUT", 1000);

// Canonical merge replanning removes components through the same exact
// acknowledgement path but hands off to HP replan removal instead of ordinary
// cleanup. The captured percentage remains untouched throughout this phase.
PROC
PROC_AESN_BeginComponentReplan((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
DB_AESN_ComponentApplication(_Combat, _Enemy, 1, "FullyCommitted")
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, _, _)
AND
DB_AESN_HpReplan(_Combat, _Enemy, 1, "RemovingComponents", _)
AND
NOT DB_AESN_HpMigrationHold(_Enemy, _)
AND
NOT DB_AESN_HpMigration(_, _Enemy, _, _, _, _, _, _, _)
THEN
PROC_AESN_SetComponentState(_Combat, _Enemy, "FullyCommitted", "ReplanningComponents");
PROC_AESN_RemoveNextComponent(_Combat, _Enemy);

PROC
PROC_AESN_BeginComponentFailureCleanup((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, _, _)
AND
DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, "Failed")
THEN
NOT DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, "Failed");
PROC_AESN_CleanupEnemy(_Enemy, _Combat);

// Public cleanup dispatch: remove only recorded component statuses first. When
// none remain, re-dispatch to the HP cleanup rule in AESN_50_Applications.
PROC
PROC_AESN_CleanupEnemy((CHARACTER)_Enemy, (GUIDSTRING)_Combat)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, _, _)
AND
NOT DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _, _)
AND
NOT DB_AESN_HpMigrationHold(_Enemy, _)
AND
NOT DB_AESN_HpMigration(_, _Enemy, _, _, _, _, _, _, _)
THEN
PROC_AESN_SetAnyComponentStateCleaning(_Combat, _Enemy);
PROC_AESN_RemoveNextComponent(_Combat, _Enemy);

PROC
PROC_AESN_SetAnyComponentStateCleaning((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, _State)
THEN
NOT DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, _State);
DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, "CleaningComponents");

PROC
PROC_AESN_RemoveNextComponent((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, "Relentless", _Status)
AND
NOT DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _, _)
THEN
DB_AESN_ComponentPendingRemove(_Combat, _Enemy, "Relentless", _Status);
RemoveStatus(_Enemy, _Status, NULL_00000000-0000-0000-0000-000000000000);
RealtimeObjectTimerLaunch(_Enemy, "AESN_COMPONENT_REMOVE_TIMEOUT", 1000);

PROC
PROC_AESN_RemoveNextComponent((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, "Relentless", _)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, "Bonus", _Status)
AND
NOT DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _, _)
THEN
DB_AESN_ComponentPendingRemove(_Combat, _Enemy, "Bonus", _Status);
RemoveStatus(_Enemy, _Status, NULL_00000000-0000-0000-0000-000000000000);
RealtimeObjectTimerLaunch(_Enemy, "AESN_COMPONENT_REMOVE_TIMEOUT", 1000);

PROC
PROC_AESN_RemoveNextComponent((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, "Relentless", _)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, "Bonus", _)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, "Action", _Status)
AND
NOT DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _, _)

PROC
PROC_AESN_RemoveNextComponent((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, "Relentless", _)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, "Bonus", _)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, "Action", _)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, "Stat", _Status)
AND
NOT DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _, _)
THEN
DB_AESN_ComponentPendingRemove(_Combat, _Enemy, "Stat", _Status);
RemoveStatus(_Enemy, _Status, NULL_00000000-0000-0000-0000-000000000000);
RealtimeObjectTimerLaunch(_Enemy, "AESN_COMPONENT_REMOVE_TIMEOUT", 1000);

PROC
PROC_AESN_RemoveNextComponent((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, _, _)
AND
DB_AESN_ComponentApplication(_Combat, _Enemy, 1, "ReplanningComponents")
AND
DB_AESN_HpReplan(_Combat, _Enemy, 1, "RemovingComponents", _)
THEN
NOT DB_AESN_ComponentApplication(_Combat, _Enemy, 1, "ReplanningComponents");
PROC_AESN_BeginHpReplanRemoval(_Combat, _Enemy);

PROC
PROC_AESN_RemoveNextComponent((GUIDSTRING)_Combat, (CHARACTER)_Enemy)
AND
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, _, _)
AND
DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, _State)
AND
_State != "ReplanningComponents"
THEN
NOT DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, _State);
DebugLog("AESN_COMPONENTS CLEANUP exactOwnedStatuses=1");
PROC_AESN_CleanupEnemy(_Enemy, _Combat);

IF
DB_AESN_ComponentApplication(_Combat, _Enemy, 1, "FullyCommitted")
AND
DB_AESN_HpReplan(_Combat, _Enemy, 1, "ApplyingNew", _CapturedPercentage)
AND
DB_AESN_MergeReplanRequired(_Combat, _Enemy)
THEN
NOT DB_AESN_MergeReplanRequired(_Combat, _Enemy);
NOT DB_AESN_HpReplan(_Combat, _Enemy, 1, "ApplyingNew", _CapturedPercentage);
DebugLog("AESN_MERGE REPLAN_COMPLETE percentageWrite=1");

IF
StatusRemoved((CHARACTER)_Enemy, _Status, _, _)
AND
DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _Kind, _Status)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, _Kind, _Status)
THEN
RealtimeObjectTimerCancel(_Enemy, "AESN_COMPONENT_REMOVE_TIMEOUT");
NOT DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _Kind, _Status);
NOT DB_AESN_EnemyComponent(_Combat, _Enemy, _Kind, _Status);
PROC_AESN_RemoveNextComponent(_Combat, _Enemy);

IF
ObjectTimerFinished((CHARACTER)_Enemy, "AESN_COMPONENT_REMOVE_TIMEOUT")
AND
DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _Kind, _Status)
THEN
NOT DB_AESN_ComponentPendingRemove(_Combat, _Enemy, _Kind, _Status);

PROC_AESN_SetAnyComponentStateCleanupFailed(_Combat, _Enemy);

THEN
NOT DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, _State);
DB_AESN_ComponentApplication(_Combat, _Enemy, _Version, "CleanupFailed");
