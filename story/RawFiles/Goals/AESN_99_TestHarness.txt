AND
GetHitpointsPercentage(_Target, _BeforePercentage)
AND
NOT DB_AESN_TestPendingApply(_Target, _, _, _)
THEN

// Restore the captured percentage exactly once for a living target after acknowledgement.
IF

AND
DB_AESN_TestPendingApply(_Target, _BeforeCurrent, _BeforeMaximum, _BeforePercentage)
AND
_BeforeCurrent > 0
AND
