// Mark-before-migrate is the ownership barrier. CombatEnded for _OldCombat can
// arrive immediately after the last native SwitchedCombat event, so both
// persistent ownership facts are inserted before any migration procedure runs.
IF
SwitchedCombat(_Object, _OldCombat, _NewCombat)
AND
_OldCombat != _NewCombat
AND
NOT DB_AESN_MergedCombat(_OldCombat, _)
THEN
DB_AESN_MergedCombat(_OldCombat, _NewCombat);
DB_AESN_CombatAlias(_OldCombat, _NewCombat);
DB_AESN_MergeInProgress(_OldCombat, _NewCombat);
DB_AESN_CombatRetired(_OldCombat);
PROC_AESN_FlattenCombatAliases(_OldCombat, _NewCombat);
PROC_AESN_MergeCombat(_OldCombat, _NewCombat);

// A later merge rewrites every earlier alias directly to the last survivor.
// Resolution therefore remains one lookup even for an arbitrary merge chain.

THEN
NOT DB_AESN_CombatAlias(_SourceCombat, _OldCombat);
DB_AESN_CombatAlias(_SourceCombat, _NewCombat);

// Reconcile the immutable policy first, then migrate component and HP state
// before eligibility facts. That ordering prevents a migrated eligible row
// from starting a second plan before its existing transaction is present.
PROC
PROC_AESN_MergeCombat((GUIDSTRING)_OldCombat, (GUIDSTRING)_NewCombat)
THEN
PROC_AESN_PreflightHpMerge(_OldCombat, _NewCombat);
PROC_AESN_ApplyPreflightedMerge(_OldCombat, _NewCombat);

PROC
PROC_AESN_ApplyPreflightedMerge((GUIDSTRING)_OldCombat, (GUIDSTRING)_NewCombat)
AND
DB_AESN_MergeInProgress(_OldCombat, _NewCombat)
AND
NOT DB_AESN_MergeHpConflict(_OldCombat, _NewCombat, _)
THEN
// Ranks are derived, not ownership. Rebuild them after the canonical ledger
// and eligibility have migrated; never carry an old selection timer forward.
PROC_AESN_CancelRelentlessSelection(_OldCombat);
PROC_AESN_CancelRelentlessSelection(_NewCombat);
PROC_AESN_DeleteRelentlessRanks(_OldCombat);
PROC_AESN_DeleteRelentlessRanks(_NewCombat);
PROC_AESN_ReconcileMergePolicy(_OldCombat, _NewCombat);
PROC_AESN_MigrateRelentlessState(_OldCombat, _NewCombat);
PROC_AESN_MigrateHpState(_OldCombat, _NewCombat);
PROC_AESN_MigrateComponentState(_OldCombat, _NewCombat);
PROC_AESN_MigrateCombatState(_OldCombat, _NewCombat);
PROC_AESN_MigrateRosterState(_OldCombat, _NewCombat);
PROC_AESN_QueueCanonicalMergeReplans(_OldCombat, _NewCombat);
NOT DB_AESN_MergeInProgress(_OldCombat, _NewCombat);
PROC_AESN_RequestRelentlessSelection(_NewCombat);
DebugLog("AESN_MERGE MIGRATION_COMPLETE markBeforeMigrate=1");

// Discarded CombatEnded is deliberately a no-op. The alias and merged marker
// remain until the final surviving owner ends and completes exact cleanup.
IF
CombatEnded(_DiscardedCombat)
AND
DB_AESN_MergedCombat(_DiscardedCombat, _)
THEN
DebugLog("AESN_MERGE CLEANUP_SUPPRESSED discardedOwner=1");

IF
CombatEnded(_SurvivingCombat)
AND
NOT DB_AESN_MergedCombat(_SurvivingCombat, _)
AND
NOT DB_AESN_CombatCleanupRequested(_SurvivingCombat)
THEN
PROC_AESN_RequestCombatCleanup(_SurvivingCombat);

PROC
PROC_AESN_RequestCombatCleanup((GUIDSTRING)_SurvivingCombat)
THEN
DB_AESN_CombatCleanupRequested(_SurvivingCombat);
PROC_AESN_CancelRelentlessSelection(_SurvivingCombat);
PROC_AESN_DispatchCombatEnemyCleanup(_SurvivingCombat);

PROC
PROC_AESN_DispatchCombatEnemyCleanup((GUIDSTRING)_Combat)
AND
DB_AESN_HpTransaction(_Combat, _Enemy, _, _, _, _, _, _, _, _)
THEN
PROC_AESN_CleanupEnemy(_Enemy, _Combat);

// A precombat enemy keeps world-owned Hardened state, so its combat may own
// only the Relentless component. Dispatch that exact component for removal too.
PROC
PROC_AESN_DispatchCombatEnemyCleanup((GUIDSTRING)_Combat)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, "Relentless", _)
AND
NOT DB_AESN_HpTransaction(_Combat, _Enemy, _, _, _, _, _, _, _, _)
THEN
PROC_AESN_CleanupEnemy(_Enemy, _Combat);

IF
DB_AESN_CombatCleanupRequested(_Combat)
AND
DB_AESN_EnemyComponent(_Combat, _Enemy, "Relentless", _)
AND
NOT DB_AESN_HpTransaction(_Combat, _Enemy, _, _, _, _, _, _, _, _)
THEN
PROC_AESN_CleanupEnemy(_Enemy, _Combat);

IF
DB_AESN_CombatCleanupRequested(_Combat)
AND
NOT DB_AESN_HpTransaction(_Combat, _, _, _, _, _, _, _, _, _)
AND
NOT DB_AESN_EnemyComponent(_Combat, _, _, _)
AND
NOT DB_AESN_RelentlessApplication(_Combat, _, _, _, "Applying")
AND
NOT DB_AESN_EnemyHpTotal(_Combat, _, _, _)
AND
NOT DB_AESN_HpTotalPending(_Combat, _, _, _, _)
AND
NOT DB_AESN_HpTotalUncertain(_Combat, _, _, _, _, _)
THEN
PROC_AESN_FinalizeCombatCleanup(_Combat);

PROC
PROC_AESN_FinalizeCombatCleanup((GUIDSTRING)_Combat)
AND
NOT DB_AESN_EnemyHpTotal(_Combat, _, _, _)
AND
NOT DB_AESN_HpTotalPending(_Combat, _, _, _, _)
AND
NOT DB_AESN_HpTotalUncertain(_Combat, _, _, _, _, _)
THEN
NOT DB_AESN_CombatCleanupRequested(_Combat);
PROC_AESN_DeleteCombatOwnedFacts(_Combat);
PROC_AESN_DeleteResolvedAliases(_Combat);
DebugLog("AESN_MERGE FINAL_CLEANUP survivingOwner=1");
