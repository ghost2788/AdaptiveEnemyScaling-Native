// Repository-only Story matrix. Production synchronization excludes this goal.
// One command runs every approved level boundary against every party size 1-12
// through the production snapshot procedures, using the host GUID as an inert
// synthetic combat key.
IF
TextEvent("AESN_TEST_SCHEMA2_POLICY_MATRIX")
AND
GetHostCharacter((CHARACTER)_Host)
AND
NOT DB_AESN_PolicyHarnessActive(_, _)
THEN
DB_AESN_PolicyHarnessActive((GUIDSTRING)_Host, 1);
PROC_AESN_RunPolicyHarnessCase((GUIDSTRING)_Host, 1);

PROC
PROC_AESN_RunPolicyHarnessCase((GUIDSTRING)_Combat, (INTEGER)_Index)
AND
DB_AESN_PolicyHarnessCase(_Index, _Level, _Size, _ExpectedTier, _ExpectedHpPercent, _ExpectedActionBudget, _ExpectedBonusBudget, _ExpectedRecipientCap)
AND
IntegerProduct(_Level, _Size, _LevelSum)
AND
NOT DB_AESN_RosterAggregate(_Combat, _, _)
AND
NOT DB_AESN_CombatSnapshotV2(_Combat, _, _, _, _, _, _, _, _, _, _, _)
THEN
DB_AESN_RosterAggregate(_Combat, _Size, _LevelSum);
PROC_AESN_FinalizeSnapshot(_Combat);
RealtimeObjectTimerLaunch((CHARACTER)_Combat, "AESN_POLICY_HARNESS_TIMEOUT", 1000);

IF
DB_AESN_PolicyHarnessActive(_Combat, _Index)
AND
DB_AESN_PolicyHarnessCase(_Index, _Level, _Size, _ExpectedTier, _ExpectedHpPercent, _ExpectedActionBudget, _ExpectedBonusBudget, _ExpectedRecipientCap)
AND
IntegerProduct(_Level, _Size, _LevelSum)
AND
DB_AESN_CombatSnapshotV2(_Combat, 2, _Size, _Size, _LevelSum, _Level, _ExpectedTier, _ExpectedHpPercent, _ExpectedActionBudget, _ExpectedBonusBudget, _ExpectedRecipientCap, "Supported")
AND
_Index < 144
AND
IntegerSum(_Index, 1, _NextIndex)
THEN
RealtimeObjectTimerCancel((CHARACTER)_Combat, "AESN_POLICY_HARNESS_TIMEOUT");
NOT DB_AESN_PolicyHarnessActive(_Combat, _Index);
PROC_AESN_DeleteCombatOwnedFacts(_Combat);
DB_AESN_PolicyHarnessActive(_Combat, _NextIndex);
PROC_AESN_RunPolicyHarnessCase(_Combat, _NextIndex);

IF
DB_AESN_PolicyHarnessActive(_Combat, 144)
AND
DB_AESN_PolicyHarnessCase(144, _Level, _Size, _ExpectedTier, _ExpectedHpPercent, _ExpectedActionBudget, _ExpectedBonusBudget, _ExpectedRecipientCap)
AND
IntegerProduct(_Level, _Size, _LevelSum)
AND
DB_AESN_CombatSnapshotV2(_Combat, 2, _Size, _Size, _LevelSum, _Level, _ExpectedTier, _ExpectedHpPercent, _ExpectedActionBudget, _ExpectedBonusBudget, _ExpectedRecipientCap, "Supported")
THEN
RealtimeObjectTimerCancel((CHARACTER)_Combat, "AESN_POLICY_HARNESS_TIMEOUT");
NOT DB_AESN_PolicyHarnessActive(_Combat, 144);
PROC_AESN_DeleteCombatOwnedFacts(_Combat);
DebugLog("AESN_POLICY_HARNESS PASS cases=144");

IF
ObjectTimerFinished((CHARACTER)_Combat, "AESN_POLICY_HARNESS_TIMEOUT")
AND
DB_AESN_PolicyHarnessActive(_Combat, _Index)
THEN
NOT DB_AESN_PolicyHarnessActive(_Combat, _Index);
PROC_AESN_DeleteCombatOwnedFacts(_Combat);
DebugLog("AESN_POLICY_HARNESS FAIL snapshot mismatch or missing row");
