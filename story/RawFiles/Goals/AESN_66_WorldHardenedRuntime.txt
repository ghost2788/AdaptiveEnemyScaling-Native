// Retrying the same debounced timer lets acknowledged HP/component work finish
// before the world snapshot is replaced.

AND

THEN

AND

AND
DB_AESN_CombatSnapshotV2(_World, 2, _, _, _, _, _Tier, _TargetHpPercent, _, _, _, "Supported")
AND
NOT DB_AESN_WorldScanOpen(_World)
AND
NOT DB_AESN_WorldMutationPending(_)
AND
NOT DB_AESN_HpApplicationHold(_World, _)
